find_package(Qt6 REQUIRED COMPONENTS Test)
find_program(BWRAP bwrap REQUIRED)

add_executable(imv-wire-test wire-test.cpp)
target_link_libraries(imv-wire-test PRIVATE imv-wire imvqml Qt6::Test)
add_dependencies(imv-wire-test imvqml-decoder)
add_test(NAME wire COMMAND imv-wire-test)
set_tests_properties(wire PROPERTIES TIMEOUT 10)

add_executable(imv-fault-child fault-child.cpp)
target_link_libraries(imv-fault-child PRIVATE imv-wire)
add_executable(imv-pool-test pool-test.cpp)
target_link_libraries(imv-pool-test PRIVATE imvqml imv-wire Qt6::Test)
target_compile_definitions(imv-pool-test PRIVATE IMV_FAULT_CHILD="$<TARGET_FILE:imv-fault-child>")
add_dependencies(imv-pool-test imv-fault-child)
add_test(NAME pool COMMAND imv-pool-test)
set_tests_properties(pool PROPERTIES TIMEOUT 18)

find_program(IMV_MAGICK magick REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(fixtures "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
add_test(NAME generate-fixtures
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate-fixtures.py" "${fixtures}" "${IMV_MAGICK}")
set_tests_properties(generate-fixtures PROPERTIES FIXTURES_SETUP images TIMEOUT 30)
add_executable(imv-job-test job-test.cpp)
target_link_libraries(imv-job-test PRIVATE imvqml Qt6::Test Qt6::Gui)
target_compile_definitions(imv-job-test PRIVATE IMV_FIXTURES="${fixtures}")
add_dependencies(imv-job-test imvqml-decoder)
add_test(NAME jobs COMMAND imv-job-test)
set_tests_properties(jobs PROPERTIES FIXTURES_REQUIRED images TIMEOUT 30)

pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
add_executable(imv-store-benchmark store-benchmark.cpp)
target_link_libraries(imv-store-benchmark PRIVATE Qt6::Gui PkgConfig::WEBP PkgConfig::ZSTD)

configure_file(relocate.cmake.in relocate.cmake @ONLY)
add_test(NAME installed-prefix COMMAND "${CMAKE_COMMAND}" -P relocate.cmake)
add_test(NAME installed-sample COMMAND "${CMAKE_COMMAND}" -DPIXELS=ON -P relocate.cmake)
set_tests_properties(installed-prefix installed-sample PROPERTIES TIMEOUT 90)
